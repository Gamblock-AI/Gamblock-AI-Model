PHP — REST API Example 
// Convert a video to MP3 using TransmuteBox API
$client = new TransmuteBox\Client('YOUR_API_KEY');

$job = $client->jobs()->create([
    'tasks' => [
        'upload-file' => [
            'operation' => 'import/upload'
        ],
        'convert-video' => [
            'operation'       => 'convert',
            'input'           => 'upload-file',
            'output_format'   => 'mp3',
            'audio_bitrate'   => 320
        ],
        'export-result' => [
            'operation' => 'export/url',
            'input'     => 'convert-video'
        ]
    ]
]);

$url = $job->getOutputUrl();
